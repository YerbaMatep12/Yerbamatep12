    function setCategory(cat) {
      activeCategory = cat;
      document.querySelectorAll('.cat-btn').forEach(btn => {
        if (btn.dataset.cat === cat) {
          btn.classList.remove('bg-mate-paper', 'text-mate-dark', 'font-semibold');
          btn.classList.add('bg-mate-dark', 'text-white', 'font-bold');
        } else {
          btn.classList.remove('bg-mate-dark', 'text-white', 'font-bold');
          btn.classList.add('bg-mate-paper', 'text-mate-dark', 'font-semibold');
        }
      });
      document.getElementById('sectionTitle').textContent = cat === 'Todos' ? 'Catálogo Completo' : cat;
      renderProducts();
    }

    function selectVariant(productId, variantIdx, event) {
      if (event) event.stopPropagation();
      selectedVariants[productId] = variantIdx;
      renderProducts();
    }

    function renderProducts() {
      const grid = document.getElementById('productsGrid');
      const emptyState = document.getElementById('emptyState');
      const searchVal = document.getElementById('searchInput').value.toLowerCase().trim();
      const sortVal = document.getElementById('sortSelect').value;

      let filtered = products.filter(p => {
        const matchCat = activeCategory === 'Todos' || p.category === activeCategory;
        const textToSearch = `${p.name} ${p.category} ${p.description} ${p.variants.map(v => v.label).join(' ')}`.toLowerCase();
        const matchSearch = textToSearch.includes(searchVal);
        return matchCat && matchSearch;
      });

      if (sortVal === 'asc') {
        filtered.sort((a, b) => a.variants[selectedVariants[a.id]].price - b.variants[selectedVariants[b.id]].price);
      } else if (sortVal === 'desc') {
        filtered.sort((a, b) => b.variants[selectedVariants[b.id]].price - a.variants[selectedVariants[a.id]].price);
      }

      document.getElementById('resultsCount').textContent = `Mostrando ${filtered.length} ${filtered.length === 1 ? 'producto disponible' : 'productos disponibles'}`;

      if (filtered.length === 0) {
        grid.innerHTML = '';
        emptyState.classList.remove('hidden');
        return;
      }

      emptyState.classList.add('hidden');
      grid.innerHTML = filtered.map(p => {
        const vIdx = selectedVariants[p.id] || 0;
        const currentVariant = p.variants[vIdx];
        const hasCustomPhoto = Boolean(customPhotos[p.id]);

        // Visual: foto subida por el usuario o SVG artesanal
        const visualHTML = hasCustomPhoto
          ? `<img src="${customPhotos[p.id]}" alt="${p.name}" class="w-full h-48 object-cover rounded-2xl" />`

               ${getProductSVG(currentVariant.visualKey)}

    function addToCart(productId, event) {
      if (event) event.stopPropagation();
      const p = products.find(item => item.id === productId);
      if (!p) return;

      const vIdx = selectedVariants[p.id] || 0;
      const v = p.variants[vIdx];
      const itemKey = `${p.id}__${v.label}`;

      const existing = cart.find(c => c.key === itemKey);
      if (existing) {
        existing.qty += 1;
      } else {
        cart.push({
          key: itemKey,
          productId: p.id,
          name: p.name,
          variantLabel: v.label,
          price: v.price,
          qty: 1
        });
      }

      updateCartUI();
      showToast(`✓ Agregado: ${p.name} (${v.label})`);
    }

    function updateQty(itemKey, delta) {
      const idx = cart.findIndex(c => c.key === itemKey);
      if (idx === -1) return;
      cart[idx].qty += delta;
      if (cart[idx].qty <= 0) {
        cart.splice(idx, 1);
      }
      updateCartUI();
    }

    function updateCartUI() {
      const totalItems = cart.reduce((acc, item) => acc + item.qty, 0);
      const totalAmount = cart.reduce((acc, item) => acc + (item.price * item.qty), 0);

      document.getElementById('headerCartCount').textContent = totalItems;
      document.getElementById('floatCartBadge').textContent = totalItems;
      document.getElementById('floatCartTotal').textContent = formatPrice(totalAmount);
      document.getElementById('drawerTotal').textContent = formatPrice(totalAmount);

      const container = document.getElementById('cartItemsContainer');
      if (cart.length === 0) {
        container.innerHTML = `
          <div class="text-center py-12 space-y-3">
            <p class="text-4xl">🧉</p>
            <p class="font-serif text-lg font-bold text-mate-dark">Tu pedido está vacío</p>
            <p class="text-xs text-mate-wood max-w-xs mx-auto">Explorá el catálogo y tocá "+ Agregar" en los mates, yerbas o accesorios que te gusten.</p>
          </div>`;
        return;
      }

      container.innerHTML = cart.map(item => `
        <div class="bg-white rounded-2xl p-3.5 border border-mate-wood/15 flex items-center justify-between gap-3">
          <div class="min-w-0 flex-1">
            <h4 class="font-serif font-bold text-sm text-mate-dark truncate">${item.name}</h4>
            <p class="text-xs text-mate-wood">Opción: ${item.variantLabel} • ${formatPrice(item.price)} c/u</p>
            <p class="text-xs font-bold text-mate-primary mt-1">${formatPrice(item.price * item.qty)}</p>
          </div>
          <div class="flex items-center gap-2 bg-mate-paper rounded-full px-2 py-1">
            <button onclick="updateQty('${item.key}', -1)" class="w-6 h-6 rounded-full bg-white text-mate-dark font-bold text-xs shadow-2xs hover:bg-mate-dark hover:text-white transition">−</button>
            <span class="text-xs font-bold w-4 text-center">${item.qty}</span>
            <button onclick="updateQty('${item.key}', 1)" class="w-6 h-6 rounded-full bg-white text-mate-dark font-bold text-xs shadow-2xs hover:bg-mate-dark hover:text-white transition">+</button>
          </div>
        </div>
      `).join('');
    }

    function buildOrderText() {
      if (cart.length === 0) return '';
      const name = document.getElementById('customerName').value.trim();
      const delivery = document.getElementById('deliveryMethod').value;
      const totalAmount = cart.reduce((acc, item) => acc + (item.price * item.qty), 0);

      let lines = [
        `¡Hola *YERBAMATEP12*! 🧉 Quiero consultar disponibilidad para este pedido:`,
        ``
      ];
      if (name) lines.push(`👤 *Nombre:* ${name}`);
      lines.push(`📦 *Modalidad:* ${delivery}`);
      lines.push(``);
      lines.push(`*Detalle del pedido:*`);

      cart.forEach(item => {
        lines.push(`• ${item.qty}x ${item.name} (${item.variantLabel}) — ${formatPrice(item.price * item.qty)}`);
      });

      lines.push(``);
      lines.push(`💰 *Total estimado:* ${formatPrice(totalAmount)}`);

    function copyOrderMessage() {
      if (cart.length === 0) {

    function triggerPhotoUpload(productId, event) {
      if (event) event.stopPropagation();

    function handlePhotoUpload(event) {
      const file = event.target.files[0];
      if (!file || !uploadingProductId) return;

      const reader = new FileReader();
      reader.onload = function(e) {
        // Comprimir imagen ligeramente para que quepa cómodo en localStorage
        const img = new Image();
        img.onload = function() {
          const canvas = document.createElement('canvas');
          const maxDim = 600;
          let w = img.width, h = img.height;
          if (w > maxDim || h > maxDim) {
            if (w > h) { h = Math.round(h * maxDim / w); w = maxDim; }
            else { w = Math.round(w * maxDim / h); h = maxDim; }
          }
          canvas.width = w;
          canvas.height = h;
          const ctx = canvas.getContext('2d');
          ctx.drawImage(img, 0, 0, w, h);
          const compressedDataUrl = canvas.toDataURL('image/jpeg', 0.82);

          customPhotos[uploadingProductId] = compressedDataUrl;
          try {
            localStorage.setItem('yerbamatep12_photos', JSON.stringify(customPhotos));
          } catch (err) {
            showToast('Foto cargada en vista actual (memoria llena para guardar)');
          }
          renderProducts();
          showToast('📸 ¡Foto actualizada con éxito!');
        };
        img.src = e.target.result;
      };
      reader.readAsDataURL(file);
      event.target.value = '';
    }

    function resetAllCustomPhotos() {
      customPhotos = {};
      try {
        localStorage.removeItem('yerbamatep12_photos');
      } catch (e) {}
      renderProducts();
      showToast('Se restauraron las ilustraciones originales');
    }
